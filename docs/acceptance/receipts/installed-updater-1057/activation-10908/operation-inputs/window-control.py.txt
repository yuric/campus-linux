def save(name, value):
    destination = OUTPUT / name
    with destination.open('x') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')


def rootpy(code):
    argv = VM + ['/usr/bin/python3', '-I', '-B', '-c', code]
    result = subprocess.run(argv, capture_output=True, text=True, timeout=8)
    if result.returncode:
        raise RuntimeError(json.dumps({'argv': argv, 'returncode': result.returncode,
                                      'stdout': result.stdout, 'stderr': result.stderr}))
    return result.stdout


def capacity():
    allocated = 0
    for directory, directories, files in os.walk(WORKTREE, followlinks=False):
        allocated += os.lstat(directory).st_blocks * 512
        for name in files:
            allocated += os.lstat(Path(directory) / name).st_blocks * 512
        for name in directories:
            path = Path(directory) / name
            if path.is_symlink():
                allocated += path.lstat().st_blocks * 512
    disk = os.stat(DISK)
    growth = max(0, disk.st_blocks * 512 - BASELINE)
    return {'observed_unix_ns': time.time_ns(), 'host_free_bytes': shutil.disk_usage('/Users/yuri').free,
            'disk_inode': disk.st_ino, 'disk_allocated_bytes': disk.st_blocks * 512,
            'stopped_baseline_allocated_bytes': BASELINE, 'new_disk_allocated_bytes': growth,
            'worktree_allocated_bytes': allocated, 'unallocated_receipt_reserve_bytes': RESERVE,
            'new_allocation_with_reserve_bytes': growth + allocated + RESERVE,
            'new_allocation_limit_bytes': LIMIT,
            'method': 'new allocated guest disk plus entire current worktree allocation plus 8 MiB reserve; original 1057 aggregate reported separately'}

def restored(state):
    properties = dict(line.split('=', 1) for line in state['state']['stdout'].splitlines() if '=' in line)
    return (state['state']['returncode'] == 0 and properties.get('LoadState') == 'loaded'
            and properties.get('ActiveState') == 'active' and properties.get('SubState') == 'running'
            and not properties.get('DropInPaths') and state['owner']['returncode'] == 0
            and state['owner']['stdout'].strip() == 'b true'
            and not any(state[key] for key in ('rule_exists', 'runtime_mask_exists', 'persistent_mask_exists')))

'''.replace('WINDOW', repr(command['authority'])).replace('LEASE', repr(LEASE)).replace('COMMAND', repr(command)).replace('GUEST', repr(GUEST))
    queue = json.loads(rootpy(queue_code))

except Exception as error:

finally:

    except Exception as error:

            or final.get('authority_summary', {}).get('cleanup') != 'restored'
            or final.get('authority_summary', {}).get('deadline_exceeded') is not False
            or not all(final.get(key) for key in ('independently_restored', 'guest_floor_after',
                                                'host_floor_after', 'allocation_after_within_limit'))):
        raise SystemExit(1)
